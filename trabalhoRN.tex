	\section{1. Como o Mapa Auto-Organizável Funciona? (Q13)}
	
	\subsection{A Intuição}
	\begin{frame}{A Metáfora do "Lençol de Borracha"}
		Para entender o SOM, imagine um espaço 3D contendo uma nuvem de pontos (nossos dados) e um \textbf{lençol de borracha quadriculado} (nossa rede neural).
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{O Processo Físico}
					\begin{itemize}
						\item Escolhemos um ponto de dado aleatório.
						\item Agarramos o \textbf{nó do lençol} mais próximo desse ponto e o puxamos até ele.
						\item Como é elástico, ele \textbf{arrasta os nós vizinhos} junto com ele.
					\end{itemize}
				\end{block}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\begin{alertblock}{Resfriamento}
					No início, o lençol é muito flexível e se move em grandes blocos (ordenação global). Com o tempo, ele enrijece e ajusta apenas detalhes (convergência local).
				\end{alertblock}
			\end{column}
		\end{columns}
	\end{frame}
	
	\subsection{A Estrutura Matemática}
	\subsection{O Algoritmo: Passo a Passo}
	
	\begin{frame}{Ciclo de Treinamento: A Estrutura e Inicialização do Mapa}
		O SOM conecta dois espaços distintos: a \textbf{grade topológica 2D} ($\mathbb{Z}^2$, que define as vizinhanças) e o \textbf{espaço de entrada} ($\mathbb{R}^D$, onde residem os dados $\mathbf{x}$ e os pesos $\mathbf{w}_j$).
		
		\vspace{0.2cm}
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{1. Configuração da Grade}
					\begin{itemize}
						\item Define-se o arranjo discreto (ex.: matriz $20 \times 20$).
						\item Cada nó $j$ da grade é um neurônio que terá um vetor de pesos $\mathbf{w}_j \in \mathbb{R}^D$.
						\item A grade 2D serve como o "tecido elástico" a ser posicionado.
					\end{itemize}
				\end{block}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\begin{block}{2. Posições Iniciais no Espaço $\mathbb{R}^D$}
					\begin{itemize}
						\item \textbf{Aleatória:} Valores ao acaso em $\mathbb{R}^D$ (mais lenta, risco de torções topológicas).
						\item \textbf{Amostral:} Sorteia instâncias reais do próprio dataset para iniciar os nós.
					\end{itemize}
				\end{block}
			\end{column}
		\end{columns}
	\end{frame}
	
	\subsection{O Algoritmo: Passo a Passo}
	\begin{frame}{Ciclo de Treinamento: Competição e Cooperação}
		A cada iteração (ou época), a rede executa etapas baseadas em princípios biológicos:
		
		\begin{enumerate}
			\item \textbf{Competição (Achar o Vencedor - BMU):}
			Apresentamos um dado $\mathbf{x}$. O neurônio $c$ com o vetor de pesos mais próximo vence (menor distância euclidiana):
			\[
			c = \arg\min_j \|\mathbf{x} - \mathbf{w}_j\|
			\]
			
			\item \textbf{Cooperação (Vizinhança):}
			O vencedor $c$ avisa seus vizinhos na grade 2D que eles também vão aprender. A força do aviso decai com a distância na grade, usando uma Gaussiana:
			\[
			h_{c, j}(t) = \exp\left( -\frac{\|\mathbf{r}_c - \mathbf{r}_j\|^2}{2\sigma^2(t)} \right)
			\]
		\end{enumerate}
	\end{frame}
	
	\begin{frame}{Ciclo de Treinamento: Adaptação}
		\begin{enumerate}
			\setcounter{enumi}{2}
			\item \textbf{Adaptação (Puxar o lençol):}
			Os pesos do vencedor e de seus vizinhos são movidos em linha reta na direção do dado $\mathbf{x}$:
			\[
			\mathbf{w}_j(t+1) = \mathbf{w}_j(t) + \eta(t) \cdot h_{c, j}(t) \cdot \big(\mathbf{x} - \mathbf{w}_j(t)\big)
			\]
			\begin{itemize}
				\item $\eta(t)$: Taxa de aprendizado (tamanho do passo).
				\item $(\mathbf{x} - \mathbf{w}_j)$: Vetor direção apontando do neurônio para o dado.
			\end{itemize}
			
			\vspace{0.3cm}
			\item \textbf{Decaimento (Resfriamento):}
			A taxa de aprendizado $\eta(t)$ e o raio de vizinhança $\sigma(t)$ diminuem ao longo do tempo. O mapa passa de uma \textit{ordenação global} grossa para um \textit{ajuste fino} local.
		\end{enumerate}
	\end{frame}
	
	\subsection{Pseudocódigo}
	\begin{frame}{Algoritmo Geral: Ciclo de Treinamento do SOM}
		\begin{columns}[T]
			\begin{column}{0.62\textwidth}
				\begin{algorithm}[H]
					\caption{Treinamento do SOM Clássico}
					\KwIn{$X \in \mathbb{R}^{n \times p}$, $\epsilon_0, \sigma_0, T_{\max}$, distâncias $d$ e $d^*$}
					\KwOut{Grade treinada de pesos $E = \{W_i\}$}
					Inicializar a grade $E$\ com pesos aleatórios $W_i$\;
					\For{$t \leftarrow 1$ \KwTo $T_{\max}$}{
						Selecionar $X^* \in X$ aleatoriamente\;
						$W^* \leftarrow \arg\min_{W \in E} d(W, X^*)$ \tcp*{\tiny Achar o BMU}
						\For{cada neurônio $W_i \in E$}{
							$W_i(t+1) \leftarrow W_i(t) + \Theta(t)\,\epsilon(t)\big(X^* - W_i(t)\big)$\;
						}
						Atualizar decaimentos:\\
						\quad $\Theta(t) = \exp\left(-\frac{d^*(W^*, W_i)}{2\sigma^2(t)}\right)$\;
						\quad $\sigma(t) = \sigma_0 \exp(-t / T_{\max})$\;
						\quad $\epsilon(t) = \epsilon_0 \exp(-t / T_{\max})$\;
					}
				\end{algorithm}
			\end{column}
			
			\begin{column}{0.36\textwidth}
				\begin{block}{Parâmetros-Chave}
					\begin{itemize}
						\item \textbf{$d$ vs. $d^*$:} $d$ opera no espaço dos dados ($\mathbb{R}^p$); $d^*$ opera na topologia da grade.
						\item \textbf{$X^*$:} Amostra estocástica sorteada a cada época/passo $t$.
						\item \textbf{$W^*$:} O neurônio vencedor (BMU).
						\item \textbf{$\Theta(t)$:} Ponderador gaussiano de vizinhança.
					\end{itemize}
				\end{block}
			\end{column}
		\end{columns}
	\end{frame}
	
	\subsection{Classificação como Rede Neural}
	\begin{frame}{Fuga do MLP: Por que ainda é uma Rede Neural?}
		Ao contrário das redes Multicamadas (MLP), o SOM \textbf{não utiliza retropropagação de erro (backpropagation) nem gradiente descendente}.
		
		Ainda assim, é legitimamente uma Rede Neural Artificial:
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{Paralelismo e Pesos}
					\begin{itemize}
						\item O conhecimento é armazenado de forma distribuída na magnitude dos \textbf{pesos sinápticos}.
					\end{itemize}
				\end{block}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\begin{block}{Inspiração Biológica Real}
					\begin{itemize}
						\item Modela a \textbf{plasticidade do córtex cerebral} (mapas somatossensoriais).
						\item Baseia-se na \textit{inibição lateral} e no aprendizado Hebbiano.
					\end{itemize}
				\end{block}
			\end{column}
		\end{columns}
	\end{frame}

		Para visualizar a topologia, treinamos um SOM com \textbf{cores aleatórias}:
		
		\begin{itemize}
			\item \textbf{Entrada ($D=3$):} Vetores com canais Vermelho, Verde e Azul $[R, G, B]$.
			\item \textbf{Saída (Grade 2D):} Matriz de $40 \times 40$ neurônios.

		\begin{exampleblock}{A Evolução do Treinamento}

				\item \textbf{Meio:} Manchas borradas agrupam tons parecidos (verdes puxando verdes).
				\item \textbf{Fim:} Um arco-íris contínuo e suave, provando que o espaço 3D foi achatado em 2D perfeitamente.
			\end{enumerate}
		\end{exampleblock}
	\end{frame}

	\section{2. Padrões sem Rótulos (Q10)}
	
	\subsection{Descoberta de Estruturas}
	\begin{frame}{Como descobrir padrões sem rótulos?}
		Sem rótulos de classe ou supervisão externa, a descoberta de padrões fundamenta-se unicamente nas **regularidades intrínsecas** do conjunto de dados:
		
		\begin{itemize}
			\item \textbf{Geometria e Proximidade:} Amostras com características comuns agrupam-se naturalmente sob neurônios vizinhos no mapa.
			\item \textbf{Densidade de Probabilidade:} O mapa aproxima a distribuição contínua dos dados — regiões mais densas ativam e moldam mais neurônios.
			\item \textbf{Visualização com Matriz-U:} Mede as distâncias euclidianas entre os pesos de neurônios vizinhos. Vales representam agrupamentos homogêneos e cumes indicam as fronteiras de separação.
		\end{itemize}
		
		\begin{block}{Conexão com o Aprendizado Não Supervisionado}
			A rede não precisa de respostas prévias; ela organiza a informação mapeando correlações estatísticas entre as variáveis de entrada.
		\end{block}
	\end{frame}

	\section{3. Redução de Dimensionalidade (Q15)}
	
	\subsection{Preservação da Informação}
	\begin{frame}{Como reduzir a dimensionalidade preservando informação?}
		Reduzir dimensões ($D \to 2$ ou $3$) exige projetar os dados mantendo a sua geometria fundamental:
		
		\begin{columns}[T]
			\begin{column}{0.48\textwidth}
				\begin{block}{Preservação Topológica}
					\begin{itemize}
						\item Se dois pontos estão próximos no espaço original $\mathbb{R}^D$, seus respectivos neurônios vencedores estarão próximos na grade 2D.
						\item Capacidade de desdobramento não-linear de variedades (\textit{manifold unrolling}).
					\end{itemize}
				\end{block}
			\end{column}
			
			\begin{column}{0.48\textwidth}
				\begin{block}{Vantagens da Projeção}
					\begin{itemize}
						\item Compressão mantendo a estrutura global de vizinhança.
						\item Facilita a inspeção visual humana de distribuições complexas em alta dimensão.
					\end{itemize}
				\end{block}
			\end{column}
		\end{columns}
	\end{frame}
	
	\subsection{Conclusão}
	\begin{frame}{Síntese e Considerações Finais}
		\begin{itemize}
			\item O **Mapa Auto-Organizável** organiza informações por competição, cooperação espacial e atualização sináptica adaptativa (Q13).
			\item É uma rede neural legítima ancorada na neurobiologia cortical e no aprendizado hebbiano, operando sem retropropagação.
			\item Identifica **agrupamentos e padrões latentes** sem necessidade de rótulos prévios (Q10).
			\item Executa uma **redução de dimensionalidade não-linear** com preservação estrita da topologia dos dados (Q15).
		\end{itemize}
